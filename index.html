const teams = [ "윤영주, 정우리", "이도혜", "김현수", "김진석, 김성욱" ]; function getDutyInfo() { const now = new Date(); const baseMon = new Date(2026, 5, 22); const currentDay = now.getDay(); const diffToMon = (currentDay === 0 ? -6 : 1 - currentDay); const thisMon = new Date(now); thisMon.setDate(now.getDate() + diffToMon); thisMon.setHours(0, 0, 0, 0); const diffWeeks = Math.round((thisMon - baseMon) / (1000 * 60 * 60 * 24 * 7)); let thisWeekIndex = diffWeeks % 4; if (thisWeekIndex < 0) thisWeekIndex += 4; let nextWeekIndex = (thisWeekIndex + 1) % 4; let afterNextWeekIndex = (thisWeekIndex + 2) % 4; document.getElementById("currentTeam").textContent = teams[thisWeekIndex]; document.getElementById("nextTeam").textContent = teams[nextWeekIndex]; document.getElementById("afterNextWeekIndex")?.textContent; document.getElementById("afterNextTeam").textContent = teams[afterNextWeekIndex]; if (currentDay === 1) { document.getElementById("currentDutyTitle").textContent = "오늘(월요일) 분리수거 당번"; } else if (currentDay === 4) { document.getElementById("currentDutyTitle").textContent = "오늘(목요일) 분리수거 당번"; } else { document.getElementById("currentDutyTitle").textContent = "이번 주 담당 조 (월/목 전담)"; } } getDutyInfo();
